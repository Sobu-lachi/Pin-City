authRouter.post('/logout', async (req, res) => {
    const refreshToken = req.cookies.refreshToken;

    try {
        // 1. Delete the token record from PostgreSQL so it can never be used to refresh again
        if (refreshToken) {
            await pool.query('DELETE FROM user_sessions WHERE token = $1', [refreshToken]);
        }

        // 2. Clear the HttpOnly cookie out of the browser completely
        res.clearCookie('refreshToken');

        res.json({ message: "Logged out from this device successfully!" });
    } catch (error) {
        res.status(500).json({ message: "Error logging out." });
    }
});

async function handleLogout() {
    try {
        // 1. Hit the backend logout endpoint (Axios automatically sends the cookie via withCredentials)
        await api.post('/logout'); 
        
        // 2. Wipe the access token out of local memory storage
        localStorage.removeItem('accessToken');

        // 3. Kick them out to the login screen
        navigate('/');
    } catch (err) {
        console.error("Logout failed:", err);
    }
}

// ThemeContext.jsx
import { createContext, useContext, useState, useEffect } from 'react';

const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light'); // Can be 'light' or 'dark'

  useEffect(() => {
    if (theme === 'dark') {
      document.body.style.backgroundColor = '#1a1a1a';
      document.body.style.color = '#ffffff';
    } else {
      document.body.style.backgroundColor = '#ffffff';
      document.body.style.color = '#000000';
    }
  }, [theme]);

  const toggleTheme = () => {
    setTheme((prevTheme) => (prevTheme === 'light' ? 'dark' : 'light'));
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export const useTheme = () => useContext(ThemeContext);


//Timed div

import React, { useState, useEffect } from 'react';

function LoginButton() {
  const [showNotification, setShowNotification] = useState(false);

  const handleLogin = () => {
    // 1. Trigger your login logic here
    // 2. Show the success notification
    setShowNotification(true);
  };

  useEffect(() => {
    // If notification is not visible, do nothing
    if (!showNotification) return;

    // Set a timer to hide the notification after 3 seconds
    const timer = setTimeout(() => {
      setShowNotification(false);
    }, 3000);

    // Clean up the timer if the component unmounts
    return () => clearTimeout(timer);
  }, [showNotification]);

  return (
    <div style={{ position: 'relative', padding: '20px' }}>
      <button onClick={handleLogin}>Log In</button>

      {showNotification && (
        <div style={{
          position: 'fixed',
          top: '20px',
          right: '20px',
          backgroundColor: '#4CAF50',
          color: 'white',
          padding: '15px',
          borderRadius: '5px',
          boxShadow: '0px 4px 6px rgba(0,0,0,0.1)'
        }}>
          Login successful!
        </div>
      )}
    </div>
  );
}

export default LoginButton;
